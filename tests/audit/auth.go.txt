package auth

import (
	"runtime"
	"testing"
)

func TestAuditAuthenticationBudget(t *testing.T) {
	a := New()
	t.Logf("NumCPU=%d GOMAXPROCS=%d CPU slots=%d aggregate Argon2 budget=%d MiB", runtime.NumCPU(), runtime.GOMAXPROCS(0), cap(a.slots), a.memoryKiB/1024)
	large := "$argon2id$v=19$m=1048576,t=64,p=1$MDEyMzQ1Njc4OWFiY2RlZg$MDEyMzQ1Njc4OWFiY2RlZjAxMjM0NTY3ODlhYmNkZWY"
	if err := ValidateHash(large); err != nil {
		t.Fatal(err)
	}
	err := ValidateHashWithMemory(large, a.memoryKiB)
	if err == nil { t.Fatal("1 GiB hash fits 128 MiB budget") }
	t.Logf("configured 1 GiB hash rejected under default budget: %v", err)
}
